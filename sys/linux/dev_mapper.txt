# Device mapper control interface (drivers/md/dm-ioctl.c) and target constructor
# parameter strings for the targets implemented in drivers/md/dm-*.c.
# See Documentation/admin-guide/device-mapper/ for the parameter formats.
#
# Underlying devices are restricted to per-proc brd ramdisks (/dev/ramN, 4MB each,
# they keep the data) and rnull devices.
# The root disk (sda/vda) must never be referenced.

# /dev/dm-N nodes appear only after a device is created, so they can't be opened
# by fixed paths (the machine check disables such openat's). Instead, create a node
# from the dev number returned by DM_DEV_CREATE/DM_DEV_STATUS (huge_encode_dev() is
# the same as new_encode_dev() expected by mknod) and open it.
resource dm_dev_number[int64]
define DM_MKNOD_MODE	(S_IFBLK | S_IRUSR | S_IWUSR)

type dm_ioctl[ID, DATA] dm_ioctl_t[ID, DATA, const[0, int32], const[0, int32]]

# struct dm_ioctl followed by the command-specific payload at data_start.
type dm_ioctl_t[ID, DATA, TARGET_COUNT, EVENT_NR] {
	version_major	const[DM_VERSION_MAJOR, int32]
# version[3] in struct dm_ioctl; check_version() requires major == DM_VERSION_MAJOR
# and minor <= the kernel's DM_VERSION_MINOR, so pass 0.
	version_minor	const[0, int32]
	version_patch	const[0, int32]
	data_size	bytesize[parent, int32]
	data_start	offsetof[data, int32]
	target_count	TARGET_COUNT
	open_count	const[0, int32]
	flags		flags[dm_ioctl_flags, int32]
	event_nr	EVENT_NR
	padding		const[0, int32]
# Covers dev (8B), name (128B), uuid (129B) and char data[7] in struct dm_ioctl.
	id		ID
	data		DATA
} [align[8]]

# DM device queues I/O indefinitely and causes close() -> sync_blockdev() to hang.
# DM_PERSISTENT_DEV_FLAG is excluded because DM_DEV_CREATE (the only ioctl reading it)
