# Arbitrary subtypes and lengths to exercise the size checks in mptcp_parse_option().
mptcp_generic_option {
	type	const[TCPOPT_MPTCP, int8]
	length	len[parent, int8]
	flags	int8:4
	subtype	int8:4
	data	array[int8, 0:36]
} [packed]

# mptcp_parse_option() requires (flags & MPTCP_CAP_FLAG_MASK) == MPTCP_CAP_HMAC_SHA256 and
# MPTCP_CAP_EXTENSIBILITY to be unset, so enumerate the valid combinations of
# MPTCP_CAP_HMAC_SHA256 with MPTCP_CAP_DENY_JOIN_ID0 and MPTCP_CAP_CHECKSUM_REQD.
mptcp_cap_flags = MPTCP_CAP_HMAC_SHA256, MPTCP_CAP_SHA256_DENY_JOIN_ID0, MPTCP_CAP_SHA256_CSUM, MPTCP_CAP_SHA256_DENY_JOIN_ID0_CSUM

define MPTCP_CAP_SHA256_DENY_JOIN_ID0	MPTCP_CAP_HMAC_SHA256 | MPTCP_CAP_DENY_JOIN_ID0
define MPTCP_CAP_SHA256_CSUM	MPTCP_CAP_HMAC_SHA256 | MPTCP_CAP_CHECKSUM_REQD
define MPTCP_CAP_SHA256_DENY_JOIN_ID0_CSUM	MPTCP_CAP_HMAC_SHA256 | MPTCP_CAP_DENY_JOIN_ID0 | MPTCP_CAP_CHECKSUM_REQD

# mptcp_parse_option() requires the DSS option size to exactly match the flags
# (TCPOLEN_MPTCP_DSS_BASE + optional ACK32/ACK64 + optional MAP32/MAP64 + optional checksum).
mptcp_dss_option [
	none		mptcp_dss_option_t[const[0x0, int8], void, void]
	ack32		mptcp_dss_option_t[const[MPTCP_DSS_HAS_ACK, int8], int32be, void]
	ack64		mptcp_dss_option_t[const[MPTCP_DSS_ACK64_FLAGS, int8], int64be, void]
	map32		mptcp_dss_option_t[flags[mptcp_dss_flags_map32, int8], void, mptcp_dss_map[int32be]]
	map64		mptcp_dss_option_t[flags[mptcp_dss_flags_map64, int8], void, mptcp_dss_map[int64be]]
	ack32_map32	mptcp_dss_option_t[flags[mptcp_dss_flags_ack32_map32, int8], int32be, mptcp_dss_map[int32be]]
	ack32_map64	mptcp_dss_option_t[flags[mptcp_dss_flags_ack32_map64, int8], int32be, mptcp_dss_map[int64be]]
	ack64_map32	mptcp_dss_option_t[flags[mptcp_dss_flags_ack64_map32, int8], int64be, mptcp_dss_map[int32be]]
	ack64_map64	mptcp_dss_option_t[flags[mptcp_dss_flags_ack64_map64, int8], int64be, mptcp_dss_map[int64be]]
] [varlen]

mptcp_dss_flags_map32 = MPTCP_DSS_HAS_MAP, MPTCP_DSS_MAP32_FIN
mptcp_dss_flags_map64 = MPTCP_DSS_MAP64, MPTCP_DSS_MAP64_FIN
mptcp_dss_flags_ack32_map32 = MPTCP_DSS_ACK32_MAP32, MPTCP_DSS_ACK32_MAP32_FIN
mptcp_dss_flags_ack32_map64 = MPTCP_DSS_ACK32_MAP64, MPTCP_DSS_ACK32_MAP64_FIN
mptcp_dss_flags_ack64_map32 = MPTCP_DSS_ACK64_MAP32, MPTCP_DSS_ACK64_MAP32_FIN
mptcp_dss_flags_ack64_map64 = MPTCP_DSS_ACK64_MAP64, MPTCP_DSS_ACK64_MAP64_FIN

define MPTCP_DSS_ACK64_FLAGS	MPTCP_DSS_HAS_ACK | MPTCP_DSS_ACK64
define MPTCP_DSS_MAP32_FIN	MPTCP_DSS_HAS_MAP | MPTCP_DSS_DATA_FIN
define MPTCP_DSS_MAP64	MPTCP_DSS_HAS_MAP | MPTCP_DSS_DSN64
define MPTCP_DSS_MAP64_FIN	MPTCP_DSS_HAS_MAP | MPTCP_DSS_DSN64 | MPTCP_DSS_DATA_FIN
define MPTCP_DSS_ACK32_MAP32	MPTCP_DSS_HAS_ACK | MPTCP_DSS_HAS_MAP
define MPTCP_DSS_ACK32_MAP32_FIN	MPTCP_DSS_HAS_ACK | MPTCP_DSS_HAS_MAP | MPTCP_DSS_DATA_FIN
define MPTCP_DSS_ACK32_MAP64	MPTCP_DSS_HAS_ACK | MPTCP_DSS_HAS_MAP | MPTCP_DSS_DSN64
define MPTCP_DSS_ACK32_MAP64_FIN	MPTCP_DSS_HAS_ACK | MPTCP_DSS_HAS_MAP | MPTCP_DSS_DSN64 | MPTCP_DSS_DATA_FIN
define MPTCP_DSS_ACK64_MAP32	MPTCP_DSS_HAS_ACK | MPTCP_DSS_ACK64 | MPTCP_DSS_HAS_MAP
define MPTCP_DSS_ACK64_MAP32_FIN	MPTCP_DSS_HAS_ACK | MPTCP_DSS_ACK64 | MPTCP_DSS_HAS_MAP | MPTCP_DSS_DATA_FIN
define MPTCP_DSS_ACK64_MAP64	MPTCP_DSS_HAS_ACK | MPTCP_DSS_ACK64 | MPTCP_DSS_HAS_MAP | MPTCP_DSS_DSN64
define MPTCP_DSS_ACK64_MAP64_FIN	MPTCP_DSS_HAS_ACK | MPTCP_DSS_ACK64 | MPTCP_DSS_HAS_MAP | MPTCP_DSS_DSN64 | MPTCP_DSS_DATA_FIN

type mptcp_dss_option_t[FLAGS, ACK, MAP] {
	type		const[TCPOPT_MPTCP, int8]
	length		len[parent, int8]
	reserved	const[0, int8:4]
	subtype		const[MPTCPOPT_DSS, int8:4]
	flags		FLAGS
	data_ack	ACK
	map		MAP
} [packed]

type mptcp_dss_map[DSN] {
	data_seq	DSN
	subflow_seq	int32be
	data_len	int16be
	csum		array[int16be, 0:1]
} [packed]

type mptcp_add_addr_option[ECHO, ADDR, PORT, HMAC] {
