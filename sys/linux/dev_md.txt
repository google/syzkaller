# /dev/mdN nodes do not exist until an array is allocated. Writing "mdN" or "md_NAME"
# to the new_array module parameter allocates /dev/mdN or /dev/md_NAME, after which
# the node can be opened. Arrays persist across test programs until stopped, so each
# proc gets a numbered array (md<proc>, freed by STOP_ARRAY or array_state=clear) and
# several named arrays (md_syz<'a'+proc>{0,1,2}; these have hold_active = UNTIL_STOP and
# are never freed in the default legacy_async_del_gendisk mode).
# The sysfs/new_array fds are not fd subtypes, so that generic write() can't be used
# to write arbitrary values (e.g. the root disk device number to md/new_dev).
resource fd_md_new_array[int32]: -1
resource fd_md[fd_block]

openat$md_new_array(fd const[AT_FDCWD], file ptr[in, string["/sys/module/md_mod/parameters/new_array"]], flags const[O_WRONLY], mode const[0]) fd_md_new_array
write$md_new_array(fd fd_md_new_array, buf ptr[in, md_array_name], len bytesize[buf]) (no_squash)

openat$md(fd const[AT_FDCWD], file ptr[in, md_dev_node], flags flags[md_open_flags], mode const[0]) fd_md (no_squash)

# Alternatively, create an md block device node (MD_MAJOR << 8, per-proc
# minor) and open it: md_probe() then allocates the array on open (create_on_open,

define MD_DEVT_BASE	(MD_MAJOR << 8)
define MD_MKNOD_MODE	(S_IFBLK | S_IRUSR | S_IWUSR)

# Member disks.
# IMPORTANT: member disks must never reference the VM root disk (sda, virtio, etc),
# because md writes superblocks to member devices. So only brd ramdisks
# (RAMDISK_MAJOR, CONFIG_BLK_DEV_RAM) are used, with per-proc minor numbers
# (2 ramdisks per proc); the major number is also enforced in neutralize.
# Loop devices are not used: /dev/loop<proc> belongs to the executor (syz_mount_image).
md_member_dev {
	major	const[RAMDISK_MAJOR, int32]
	minor	proc[0, 2, int32]
}

# Encoded dev_t (new_encode_dev(), i.e. major << 8 | minor for small numbers)
# passed as the ioctl argument of HOT_ADD_DISK, HOT_REMOVE_DISK, SET_DISK_FAULTY.
define RAM_DEVT_BASE	(RAMDISK_MAJOR << 8)
type md_ram_devt proc[RAM_DEVT_BASE, 2, intptr]

mdu_version_t {
	major		int32
	minor		int32
	patchlevel	int32
}

mdu_array_info_t {
# Only used when raid_disks == 0 (assembling from existing superblocks):
# index into super_types (0 = 0.90, 1 = 1.x).
	major_version	int32[0:1]
	minor_version	flags[md_minor_versions, int32]
	patch_version	const[0, int32]
	ctime		int32
	level		flags[md_levels, int32]
# Component size in KiB (ramdisks are 4 MiB).
	size		int32[0:4096]
	nr_disks	const[0, int32]
# Each proc has only 2 member disks, so larger arrays can only run degraded (or not at all).
	raid_disks	int32[0:4]
	md_minor	const[0, int32]
	not_persistent	bool32
	utime		const[0, int32]
	state		flags[md_array_state, int32]
	active_disks	const[0, int32]
	working_disks	const[0, int32]
	failed_disks	const[0, int32]
	spare_disks	const[0, int32]
	layout		flags[md_layouts, int32]
	chunk_size	flags[md_chunk_sizes, int32]
}

mdu_disk_info_t {
	number		int32[0:MD_SB_DISKS]
# major and minor in struct mdu_disk_info_s.
	dev		md_member_dev
	raid_disk	int32[-1:3]
	state		flags[md_disk_state, int32]
}

# get_disk_info() only reads number and overwrites the rest.
mdu_disk_info_get_t {

	major		int32	(out)
	minor		int32	(out)
	raid_disk	int32	(out)
	state		int32	(out)
}

# raid0: RAID0_ORIG_LAYOUT, RAID0_ALT_MULTIZONE_LAYOUT (drivers/md/raid0.h).
# raid4/5/6: ALGORITHM_* (drivers/md/raid5.h).
# raid10: near_copies | (far_copies << 8) | (far_offset << 16) | (version << 17) (drivers/md/raid10.c:setup_geo).
