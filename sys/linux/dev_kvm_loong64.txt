# kvm_syz_vm is a VM handler used by syzos-related pseudo-syscalls. It is actually an opaque pointer under the hood.
resource kvm_syz_vm$loong64[int64]

# Map the given memory into the VM and set up syzos there.
# vma[256] * 16KiB target pages ~= 4MiB, matching the absolute guest memory size used by other arches.

# Create a VCPU inside a kvm_syz_vm VM.
# Prohibit flattening the input arguments, so that it is easier to reason about them.

kvm_vcpu_attr_loong64_cpucfg_feature {
	flags	const[0, int32]
# KVM_LOONGARCH_VCPU_CPUCFG from arch/loongarch/include/uapi/asm/kvm.h.
	group	const[0, int32]
# CPUCFG_KVM_FEATURE from arch/loongarch/include/uapi/asm/kvm_para.h.
	attr	const[0x40000004, int64]
	addr	ptr64[inout, flags[kvm_loong64_cpucfg_features, int64]]
}

# KVM_FEATURE_* values are feature indexes in the UAPI; the device attribute
# takes the corresponding bitmask.
kvm_loong64_cpucfg_features = 0x2, 0x4, 0x8, 0x1000000, 0x2000000

]
